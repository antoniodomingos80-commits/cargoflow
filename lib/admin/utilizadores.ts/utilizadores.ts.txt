'use server';

import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function listarUtilizadores() {
  try {
    const { data, error } = await supabase
      .from('users')
      .select('id, email, full_name, phone, verification_status, banned, created_at')
      .order('created_at', { ascending: false });

    if (error) throw error;
    return data || [];
  } catch (error) {
    console.error('Erro ao listar utilizadores:', error);
    throw error;
  }
}

export async function suspenderUtilizador(id: string) {
  try {
    const { error } = await supabase
      .from('users')
      .update({
        banned: true,
        ban_reason: 'Suspenso manualmente',
        banned_at: new Date().toISOString(),
      })
      .eq('id', id);

    if (error) throw error;
    return { success: true };
  } catch (error) {
    console.error('Erro ao suspender utilizador:', error);
    throw error;
  }
}

export async function ativarUtilizador(id: string) {
  try {
    const { error } = await supabase
      .from('users')
      .update({
        banned: false,
        ban_reason: null,
        banned_at: null,
      })
      .eq('id', id);

    if (error) throw error;
    return { success: true };
  } catch (error) {
    console.error('Erro ao ativar utilizador:', error);
    throw error;
  }
}
