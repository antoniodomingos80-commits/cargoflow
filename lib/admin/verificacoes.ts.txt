'use server';

import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function listarVerificacoes() {
  try {
    const { data, error } = await supabase
      .from('users')
      .select('id, email, full_name, verification_status')
      .eq('verification_status', 'PENDENTE')
      .order('created_at', { ascending: false });

    if (error) throw error;
    return data || [];
  } catch (error) {
    console.error('Erro ao listar verificações:', error);
    throw error;
  }
}

export async function aprovarUtilizador(id: string) {
  try {
    const { error } = await supabase
      .from('users')
      .update({
        verification_status: 'APPROVED',
        verified_at: new Date().toISOString(),
      })
      .eq('id', id);

    if (error) throw error;
    return { success: true };
  } catch (error) {
    console.error('Erro ao aprovar:', error);
    throw error;
  }
}

export async function rejeitarUtilizador(id: string, motivo: string) {
  try {
    const { error } = await supabase
      .from('users')
      .update({
        verification_status: 'REJECTED',
        rejection_reason: motivo,
        rejected_at: new Date().toISOString(),
      })
      .eq('id', id);

    if (error) throw error;
    return { success: true };
  } catch (error) {
    console.error('Erro ao rejeitar:', error);
    throw error;
  }
}
